    participants: list[str]  # sorted alphabetically by agent_id; deterministic ordering

    events: list[Event],
    room: Room,
    *,
    snapshot_at: TimestampMs,

  ```python
  # agents/snapshot_dispatch.py (clock-injection per ADR-014 D1 + D6)
  def produce_and_emit_snapshot(
      agent: Agent,
      room: Room,
      current_event: Event,
      event_store: EventStore,
      snapshot_events: list[Event],
  ) -> SnapshotDispatchResult:
      snapshot_at = now_ms()  # caller owns wall-clock impurity per ADR-014 D1
      snapshot = produce_snapshot(snapshot_events, room, snapshot_at=snapshot_at)
      meta = SummarySnapshotMeta(**snapshot.model_dump())
      event_store.append(Event(
          id=generate_event_id(), room_id=room.id, type=EventType.SUMMARY_SNAPSHOT,
          author=build_author(agent.id), timestamp=snapshot_at,
          content="", metadata=meta.model_dump(),
      ))
      return SnapshotDispatchResult(snapshot=snapshot, appended=True)
  ```
